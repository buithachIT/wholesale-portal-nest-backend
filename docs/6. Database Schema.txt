// Trạng thái của các entity có thể bị tạm dừng.
// Được dùng cho users, shipping_carriers, truck_types...
Enum entity_status {
  active    [note: "Entity đang hoạt động"]
  suspended [note: "Entity đang tạm ngưng hoạt động"]
}


// Vòng đời của một đơn hàng.
Enum order_status {
  pending        [note: "Đơn vừa được tạo, đang chờ xử lý/phê duyệt"]
  approved       [note: "Đơn đã được hệ thống/admin phê duyệt đầy đủ"]
  approved_local [note: "Đơn đã được hệ thống nội bộ duyệt nhưng chưa đồng bộ thành công với nhà vận chuyển"]
  shipping       [note: "Đơn đang trong quá trình vận chuyển"]
  delivered      [note: "Đơn đã giao hàng thành công"]
  cancelled      [note: "Đơn đã bị hủy"]
}


// Phương thức thanh toán mà doanh nghiệp lựa chọn khi đặt hàng.
Enum payment_method_type {
  credit        [note: "Mua hàng theo hạn mức công nợ được cấp"]
  bank_transfer [note: "Thanh toán bằng chuyển khoản ngân hàng"]
}


// Trạng thái thanh toán tổng thể của một đơn hàng.
// Tách riêng khỏi order_status vì vòng đời đơn và dòng tiền là hai vấn đề khác nhau.
Enum order_payment_status {
  unpaid  [note: "Đơn chưa được thanh toán"]
  partial [note: "Đơn mới được thanh toán một phần, vẫn còn công nợ"]
  paid    [note: "Đơn đã được thanh toán đủ"]
}


// Trạng thái của một khoản tiền thực tế được khai báo/chuyển khoản.
Enum payment_status_type {
  pending  [note: "Đại lý báo đã chuyển khoản, đang chờ kế toán kiểm tra tài khoản ngân hàng"]
  approved [note: "Kế toán đã xác nhận tiền thực tế đã về tài khoản"]
  rejected [note: "Khoản thanh toán bị từ chối do không tìm thấy giao dịch hoặc thông tin không hợp lệ"]
}


// Loại biến động hạn mức công nợ.
Enum credit_transaction_type {
  credit_limit_assigned [note: "Quản lý cấp hoặc điều chỉnh hạn mức công nợ cho doanh nghiệp"]
  credit_reserved       [note: "Đơn hàng trả sau được chốt và giữ lại một phần hạn mức công nợ"]
  credit_released       [note: "Đơn hàng bị hủy nên phần hạn mức đã giữ được trả lại"]
  debt_payment          [note: "Khoản thanh toán được dùng để giảm công nợ"]
  adjustment            [note: "Kế toán/quản trị viên điều chỉnh công nợ thủ công"]
}


// Loại biến động số dư tiền dư/tiền ứng trước.
Enum wallet_transaction_type {
  overpayment    [note: "Khoản tiền thanh toán vượt số tiền cần cấn trừ"]
  wallet_payment [note: "Sử dụng tiền trong wallet để thanh toán đơn hàng"]
  refund         [note: "Hoàn tiền cho doanh nghiệp"]
  adjustment     [note: "Kế toán điều chỉnh số dư wallet thủ công"]
}


// Cách tính discount của pricing tier.
// Hiện tại hệ thống chỉ hỗ trợ giảm theo phần trăm.
Enum pricing_tier_type {
  percentage_discount [note: "Giảm giá theo phần trăm trên giá gốc"]
}


// Các phân khúc xe được hệ thống hỗ trợ.
Enum truck_type_size {
  van_500kg   [note: "Xe van 500kg, sức chứa tối đa khoảng 2.5 CBM"]
  small_1_5t  [note: "Xe tải nhỏ 1.5 tấn, sức chứa tối đa khoảng 8 CBM"]
  medium_2_5t [note: "Xe tải trung 2.5 tấn, sức chứa tối đa khoảng 14 CBM"]
  large_5t    [note: "Xe tải lớn 5 tấn, sức chứa tối đa khoảng 24 CBM"]
  heavy_15t   [note: "Xe tải nặng 15 tấn, sức chứa tối đa khoảng 60 CBM"]
}


// Trạng thái của từng chuyến giao hàng.
Enum delivery_status_type {
  assigned   [note: "Đã phân công nhà xe/xe/tài xế nhưng chưa bắt đầu chạy"]
  in_transit [note: "Xe đã bắt đầu vận chuyển"]
  completed  [note: "Chuyến xe đã giao hàng hoàn tất"]
}


// Trạng thái xử lý webhook từ nhà vận chuyển.
Enum webhook_status_type {
  received  [note: "Webhook đã nhận nhưng chưa xử lý"]
  processed [note: "Webhook đã được xử lý thành công"]
  failed    [note: "Webhook xử lý thất bại, cần kiểm tra hoặc retry"]
}

// Loại notification.

  order_cancelled        [note: "Đơn hàng đã bị hủy"]

  payment_submitted      [note: "Có khoản thanh toán mới được khai báo"]
  payment_approved       [note: "Khoản thanh toán đã được xác nhận"]
  payment_rejected       [note: "Khoản thanh toán bị từ chối"]

  credit_updated         [note: "Hạn mức công nợ của doanh nghiệp được thay đổi"]

  logistics_assigned     [note: "Đơn hàng đã được phân công chuyến vận chuyển"]
  logistics_in_transit   [note: "Chuyến vận chuyển đã bắt đầu"]
  logistics_completed    [note: "Chuyến vận chuyển đã hoàn tất"]

  system                 [note: "Thông báo chung từ hệ thống"]

  logistics
  company
  credit_transaction
  wallet_transaction
  system
}

Table companies {
  id integer [pk, increment, note: "ID duy nhất của doanh nghiệp/đại lý"]

  name varchar(150) [
    not null,
    note: "Tên pháp lý/tên giao dịch của doanh nghiệp mua hàng"
  ]

  phone varchar(20) [
    note: "Số điện thoại liên hệ của doanh nghiệp"
  ]

  credit_limit bigint [
    not null,
    default: 0,
    note: "Hạn mức công nợ tối đa được super_admin cấp cho doanh nghiệp, đơn vị VNĐ"
  ]

  remaining_credit bigint [
    not null,
    default: 0,
    note: "Hạn mức công nợ hiện còn có thể sử dụng để tạo đơn trả sau, đơn vị VNĐ"
  ]

  wallet_balance bigint [
    not null,
    default: 0,
    note: "Số tiền dư/tiền ứng trước của doanh nghiệp có thể dùng cho các đơn hàng tiếp theo, đơn vị VNĐ"
  ]

  created_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm tạo doanh nghiệp"
  ]

  updated_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm cập nhật thông tin doanh nghiệp gần nhất"
  ]

  Indexes {
    name [name: "idx_companies_name"]
  }
}


// ============================================================
// 3. USERS
// ============================================================

Table users {
  id integer [pk, increment, note: "ID duy nhất của người dùng"]

  company_id integer [
    note: "Doanh nghiệp mà user thuộc về; NULL nếu đây là user nội bộ của bên bán"
  ]

  email varchar(255) [
    unique,
    not null,
    note: "Email dùng để đăng nhập hệ thống"
  ]

  password varchar(255) [
    not null,
    note: "Mật khẩu đã được hash, không lưu plaintext"
  ]

  full_name varchar(150) [
    not null,
    note: "Họ tên người dùng"
  ]

  status entity_status [
    not null,
    default: `entity_status::active`,
    note: "Trạng thái tài khoản dùng để kiểm tra lại khi user đang thao tác"
  ]

  created_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm tạo tài khoản"
  ]

  updated_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm cập nhật tài khoản gần nhất"
  ]

  Indexes {
    company_id [name: "idx_users_company"]
    status [name: "idx_users_status"]
  }
}


// ============================================================
// REFRESH TOKENS
// ============================================================

Table refresh_tokens {
  id bigint [
    pk,
    increment,
    note: "ID duy nhất của refresh token"
  ]

  user_id integer [
    not null,
    note: "User sở hữu refresh token"
  ]

  token_hash varchar(255) [
    not null,
    unique,
    note: "Hash của refresh token; không lưu raw refresh token"
  ]

  expires_at timestamp [
    not null,
    note: "Thời điểm refresh token hết hạn"
  ]

  revoked_at timestamp [
    note: "Thời điểm refresh token bị thu hồi; NULL nếu token vẫn còn hiệu lực"
  ]

  replaced_by_token_id bigint [
    note: "ID của refresh token mới thay thế token hiện tại khi thực hiện token rotation"
  ]

  created_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm tạo refresh token"
  ]

  updated_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm cập nhật refresh token gần nhất"
  ]

  Indexes {
    user_id [
      name: "idx_refresh_tokens_user"
    ]

    expires_at [
      name: "idx_refresh_tokens_expires_at"
    ]

    revoked_at [
      name: "idx_refresh_tokens_revoked_at"
    ]

    (user_id, revoked_at) [
      name: "idx_refresh_tokens_user_revoked"
    ]
  }
}

Table roles {
  id integer [pk, increment, note: "ID của role"]

  name varchar(100) [
    unique,
    not null,
    note: "Tên role, ví dụ SUPER_ADMIN, ADMIN, ACCOUNTANT, PURCHASER"
  ]

  description varchar(255) [
    note: "Mô tả mục đích của role"
  ]
}


Table permissions {
  id integer [pk, increment, note: "ID của permission"]

  code varchar(100) [
    unique,
    not null,
    note: "Mã quyền, ví dụ ORDER_CREATE, ORDER_APPROVE, PAYMENT_APPROVE, CREDIT_UPDATE"
  ]

  description varchar(255) [
    note: "Mô tả quyền"
  ]
}


Table user_roles {
  user_id integer [
    not null,
    note: "User được gán role"
  ]

  role_id integer [
    not null,
    note: "Role được gán cho user"
  ]

  Indexes {
    (user_id, role_id) [pk]
    role_id [name: "idx_user_roles_role"]
  }
}


Table role_permissions {
  role_id integer [
    not null,
    note: "Role sở hữu permission"
  ]

  permission_id integer [
    not null,
    note: "Permission được cấp cho role"
  ]

  Indexes {
    (role_id, permission_id) [pk]
    permission_id [name: "idx_role_permissions_permission"]
  }
}

    company_id [name: "idx_orders_company"]
    user_id [name: "idx_orders_user"]
    status [name: "idx_orders_status"]
    payment_status [name: "idx_orders_payment_status"]
    created_at [name: "idx_orders_created_at"]
  }
}

// ============================================================
// 12. LOGISTICS ITEMS
// ============================================================

// Một logistics có thể chỉ chở một phần hàng của order.
// logistics_items dùng để xác định chính xác những order_item nào
// được phân bổ vào chuyến xe nào.
//
// Ví dụ:
// Order #100 có:
//   - Gạo 25kg: 100 thùng
//   - Dầu ăn: 50 thùng
//
// Hệ thống có thể tạo:
//   Logistics #1:
//     - Gạo 25kg: 70 thùng
//     - Dầu ăn: 50 thùng
//
//   Logistics #2:
//     - Gạo 25kg: 30 thùng
//
// Điều này cho phép một order được chia thành nhiều chuyến xe
// và một order_item cũng có thể được chia cho nhiều logistics.


Table logistics_items {
  id integer [
    pk,
    increment,
    note: "ID dòng hàng được phân bổ vào chuyến logistics"
  ]

  logistics_id integer [
    not null,
    note: "Chuyến logistics chứa phần hàng này"
  ]

  order_item_id integer [
    not null,
    note: "Dòng sản phẩm gốc trong order được phân bổ"
  ]

  quantity integer [
    not null,
    note: "Số lượng thùng của order_item được phân bổ vào chuyến này"
  ]

  planned_cbm decimal(12,4) [
    not null,
    note: "Tổng CBM của phần hàng được phân bổ vào chuyến này"
  ]

  planned_weight_kg decimal(12,2) [
    not null,
    note: "Tổng khối lượng của phần hàng được phân bổ vào chuyến này, đơn vị kg"
  ]

  Indexes {
    logistics_id [name: "idx_logistics_items_logistics"]
    order_item_id [name: "idx_logistics_items_order_item"]

    (logistics_id, order_item_id) [
      unique,
      name: "uq_logistics_items_logistics_order_item"
    ]
  }
}

// ============================================================
// 18. NOTIFICATIONS
// ============================================================

// Notification được tạo cho một user cụ thể.
//
// Ví dụ:
//
// - Admin duyệt Order #100
//   → notification cho user đã tạo Order #100
//
// - Customer tạo Payment #20
//   → notification cho ACCOUNTANT
//
// - Carrier cập nhật Logistics #15 → DELIVERED
//   → notification cho user của Order tương ứng.
//
// reference_type + reference_id dùng để liên kết notification
// tới entity nghiệp vụ tương ứng.
//
// Không dùng FK trực tiếp cho reference_id vì một notification
// có thể tham chiếu order, payment, logistics, company...

// ============================================================
// 18. NOTIFICATIONS
// ============================================================

Table notifications {
  id bigint [
    pk,
    increment,
    note: "ID duy nhất của notification"
  ]

  user_id integer [
    not null,
    note: "User nhận notification"
  ]

  actor_user_id integer [
    note: "User thực hiện hành động tạo ra notification; NULL nếu notification được tạo tự động bởi hệ thống"
  ]

  type notification_type [
    not null,
    note: "Loại notification dùng để frontend/backend xác định nghiệp vụ và cách hiển thị"
  ]

  title varchar(200) [
    not null,
    note: "Tiêu đề notification"
  ]

  message varchar(1000) [
    not null,
    note: "Nội dung notification"
  ]

  reference_type notification_reference_type [
    note: "Loại entity mà notification tham chiếu tới, ví dụ ORDER, PAYMENT, LOGISTICS"
  ]

  reference_id bigint [
    note: "ID của entity được tham chiếu; không dùng FK vì reference có thể thuộc nhiều entity khác nhau"
  ]

  action_url varchar(500) [
    note: "URL/route frontend dùng để điều hướng user tới màn hình chi tiết liên quan khi click notification"
  ]

  read_at timestamp [
    note: "Thời điểm user đọc notification; NULL nghĩa là chưa đọc"
  ]

  created_at timestamp [
    not null,
    default: `now()`,
    note: "Thời điểm notification được tạo"
  ]

  Indexes {
    user_id [
      name: "idx_notifications_user"
    ]

    type [
      name: "idx_notifications_type"
    ]

    (user_id, read_at) [
      name: "idx_notifications_user_read_at"
    ]

    (user_id, created_at) [
      name: "idx_notifications_user_created_at"
    ]

    (reference_type, reference_id) [
      name: "idx_notifications_reference"
    ]
  }
}
